/// Uygulamanın ThemeData'sı. `MaterialApp(theme: OmvilaTema.acik(), darkTheme: OmvilaTema.koyu())`
abstract final class OmvilaTema {
  static ThemeData acik() => _tema(OmvilaRenkler.acik, Brightness.light);
  static ThemeData koyu() => _tema(OmvilaRenkler.koyu, Brightness.dark);

  static ThemeData _tema(OmvilaRenkler r, Brightness b) {
    final scheme = ColorScheme(
      brightness: b,
      primary: r.marka, onPrimary: r.markaUstu,
      primaryContainer: r.zeminSecili, onPrimaryContainer: r.metinMarka,
      secondary: r.vurgu, onSecondary: r.vurguUstu,
      error: r.hata, onError: r.markaUstu,
      surface: r.zeminYukseltilmis, onSurface: r.metin,
      onSurfaceVariant: r.metinIkincil,
      surfaceContainerHighest: r.zeminIkincil,
      outline: r.cizgiGuclu, outlineVariant: r.cizgi,
    );
    final kose = (double k) => RoundedRectangleBorder(borderRadius: BorderRadius.circular(k));
    return ThemeData(
      useMaterial3: true,
      brightness: b,
      colorScheme: scheme,
      fontFamily: OmvilaYazi.aile,
      scaffoldBackgroundColor: r.zemin,
      extensions: [r],
      textTheme: TextTheme(
        displaySmall: OmvilaYazi.buyukBaslik, headlineSmall: OmvilaYazi.baslik1,
        titleLarge: OmvilaYazi.baslik2, titleMedium: OmvilaYazi.baslik3,
        bodyLarge: OmvilaYazi.govde, bodyMedium: OmvilaYazi.ikincil,
        labelLarge: OmvilaYazi.etiket, labelMedium: OmvilaYazi.etiketKucuk, labelSmall: OmvilaYazi.aciklama,
      ).apply(bodyColor: r.metin, displayColor: r.metin),
      appBarTheme: AppBarTheme(
        backgroundColor: r.zemin, foregroundColor: r.metin, elevation: 0, scrolledUnderElevation: 0,
        centerTitle: true, toolbarHeight: OmvilaBoyut.ustBar, titleTextStyle: OmvilaYazi.ustBarBasligi.copyWith(color: r.metin),
      ),
      filledButtonTheme: FilledButtonThemeData(style: FilledButton.styleFrom(
        backgroundColor: r.marka, foregroundColor: r.markaUstu, minimumSize: const Size(64, OmvilaBoyut.kontrolMd),
        textStyle: OmvilaYazi.etiket, shape: kose(OmvilaKose.koseMd),
      )),
      outlinedButtonTheme: OutlinedButtonThemeData(style: OutlinedButton.styleFrom(
        foregroundColor: r.metin, side: BorderSide(color: r.cizgiGuclu), minimumSize: const Size(64, OmvilaBoyut.kontrolMd),
        textStyle: OmvilaYazi.etiket, shape: kose(OmvilaKose.koseMd),
      )),
      textButtonTheme: TextButtonThemeData(style: TextButton.styleFrom(
        foregroundColor: r.metinMarka, minimumSize: const Size(44, OmvilaBoyut.dokunmaMin), textStyle: OmvilaYazi.etiket,
      )),
      inputDecorationTheme: InputDecorationTheme(
        filled: true, fillColor: r.zeminYukseltilmis,
        contentPadding: const EdgeInsets.symmetric(horizontal: OmvilaBosluk.bosluk4, vertical: 14),
        border: OutlineInputBorder(borderRadius: BorderRadius.circular(OmvilaKose.koseMd), borderSide: BorderSide(color: r.cizgiGuclu)),
        enabledBorder: OutlineInputBorder(borderRadius: BorderRadius.circular(OmvilaKose.koseMd), borderSide: BorderSide(color: r.cizgiGuclu)),
        focusedBorder: OutlineInputBorder(borderRadius: BorderRadius.circular(OmvilaKose.koseMd), borderSide: BorderSide(color: r.odak, width: 2)),
        errorBorder: OutlineInputBorder(borderRadius: BorderRadius.circular(OmvilaKose.koseMd), borderSide: BorderSide(color: r.hata)),
        labelStyle: OmvilaYazi.ikincil.copyWith(color: r.metinIkincil), hintStyle: OmvilaYazi.govde.copyWith(color: r.metinIkincil),
      ),
      cardTheme: CardThemeData(color: r.zeminYukseltilmis, elevation: 0, margin: EdgeInsets.zero,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(OmvilaKose.koseLg), side: BorderSide(color: r.cizgi))),
      navigationBarTheme: NavigationBarThemeData(
        backgroundColor: r.tabbarZemin, height: OmvilaBoyut.tabbar, indicatorColor: r.zeminSecili, elevation: 0,
        iconTheme: WidgetStateProperty.resolveWith((s) => IconThemeData(color: s.contains(WidgetState.selected) ? r.tabbarAktif : r.tabbarPasif, size: 24)),
        labelTextStyle: WidgetStateProperty.resolveWith((s) => OmvilaYazi.sekme.copyWith(color: s.contains(WidgetState.selected) ? r.tabbarAktif : r.tabbarPasif)),
      ),
      bottomSheetTheme: BottomSheetThemeData(backgroundColor: r.zeminYukseltilmis, showDragHandle: true, dragHandleColor: r.tutamak,
        shape: const RoundedRectangleBorder(borderRadius: BorderRadius.vertical(top: Radius.circular(OmvilaKose.koseXl)))),
      dialogTheme: DialogThemeData(backgroundColor: r.zeminYukseltilmis, shape: kose(OmvilaKose.koseXl)),
      chipTheme: ChipThemeData(shape: kose(OmvilaKose.koseTam), side: BorderSide(color: r.cizgiGuclu), selectedColor: r.zeminSecili,
        labelStyle: OmvilaYazi.etiketKucuk.copyWith(color: r.metin)),
      snackBarTheme: SnackBarThemeData(behavior: SnackBarBehavior.floating, backgroundColor: OmvilaRampa.petrol950, contentTextStyle: OmvilaYazi.ikincil.copyWith(color: OmvilaRampa.kum50),
        shape: kose(OmvilaKose.koseMd)),
      dividerTheme: DividerThemeData(color: r.cizgi, thickness: 1, space: 1),
      switchTheme: SwitchThemeData(trackColor: WidgetStateProperty.resolveWith((s) => s.contains(WidgetState.selected) ? r.marka : r.cizgiGuclu),
        thumbColor: const WidgetStatePropertyAll(Colors.white), trackOutlineColor: const WidgetStatePropertyAll(Colors.transparent)),
    );
  }
}
